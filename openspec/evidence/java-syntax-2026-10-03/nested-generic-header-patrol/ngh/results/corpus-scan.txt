```sh
# scan_corpus.sh <out.sha>（每腿一次；leg1 = stash 基线构建，leg2 = 本片构建）
#!/bin/bash
set -u
OUT="$1"; ROOT="<repo root>"; BIN="$ROOT/target/release/examples/class_source_file"
cd "$ROOT" || exit 1
: > "$OUT"
while IFS= read -r class; do
  TEXT="$("$BIN" "$class" 2>/dev/null)"; CODE=$?
  if [ $CODE -eq 0 ]; then
    SHA=$(printf '%s' "$TEXT" | shasum -a 256 | cut -d' ' -f1); echo "OK $SHA $class" >> "$OUT"
  else
    echo "EXIT$CODE - $class" >> "$OUT"
  fi
done < <(find tests/fixtures fuzz/corpus openspec/evidence -name '*.class' -type f \
  -not -path '*/nested-generic-header-patrol/ngh/*' | sort)
```
